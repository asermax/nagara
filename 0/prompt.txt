  { selector: "blockquote", role: "unit" },
  { selector: "h1, h2, h3, h4", role: "unit" },

  { selector: "figure", role: "unit" },

  { selector: "a", role: "inline" },
  { selector: "strong", role: "inline" },
  { selector: "figcaption", role: "inline" },

export function extract($, toMarkdown) {
  const title = $("main &gt; h1").first().text().trim() || ($("meta[property='og:title']").attr("content") || "").trim();
  const units = [];
  const container = $("article");
  container.children().each((i, el) =&gt; {
    const node = $(el);
    const tag = el.tagName ? el.tagName.toLowerCase() : "";
    if (tag === "hr" || tag === "dl" || node.is(".footnotes")) return;
    if (tag === "pre") {
      units.push({ type: "code", display: toMarkdown(el), element: el });
      return;
    }
    if (tag === "figure") {
      const img = node.find("img").first();
      if (img.length) {
        units.push({ type: "image", src: img.attr("src") || "", alt: img.attr("alt") || "", element: el });
      }
      return;
    }
    if (/^h[1-4]$/.test(tag) &amp;&amp; node.text().trim() === title) return;
    if (["p", "ul", "ol", "blockquote", "h1", "h2", "h3", "h4"].includes(tag)) {
      const prepared = node.clone();
      prepared.find("sup:has(a.footnote-ref)").remove();
      const display = toMarkdown(prepared);
      if (display &amp;&amp; display.trim()) units.push({ type: "paragraph", display, element: el });
      return;
    }
    const display = toMarkdown(el);
    if (display &amp;&amp; display.trim()) units.push({ type: "paragraph", display, element: el });
  });
  return { title, units };
}
